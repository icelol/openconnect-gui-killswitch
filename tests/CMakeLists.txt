# Exercise the shipped main window with an injected PF runner. No live firewall,
# VPN socket, credentials, or normal user settings are used by these tests.
get_target_property(gui_sources ${PROJECT_NAME} SOURCES)
set(gui_test_sources)
foreach(source IN LISTS gui_sources)
    if(NOT source STREQUAL "main.cpp")
        if(source STREQUAL "timestamp.cpp")
            list(APPEND gui_test_sources "${PROJECT_BINARY_DIR}/src/timestamp.cpp")
        elseif(IS_ABSOLUTE "${source}")
            list(APPEND gui_test_sources "${source}")
        else()
            list(APPEND gui_test_sources "${PROJECT_SOURCE_DIR}/src/${source}")
        endif()
    endif()
endforeach()
get_target_property(gui_libraries ${PROJECT_NAME} LINK_LIBRARIES)
set_source_files_properties("${PROJECT_BINARY_DIR}/src/timestamp.cpp" PROPERTIES GENERATED TRUE)
add_executable(killswitch-ui-tests KillSwitchUiTest.cpp ${gui_test_sources})
add_dependencies(killswitch-ui-tests build_timestamp git_revision)
target_include_directories(killswitch-ui-tests PRIVATE ../src ${PROJECT_BINARY_DIR}/src)
target_link_libraries(killswitch-ui-tests ${gui_libraries} Qt6::Test)
set_target_properties(killswitch-ui-tests PROPERTIES AUTOUIC_SEARCH_PATHS "${PROJECT_SOURCE_DIR}/src/dialog")
add_test(NAME killswitch-ui COMMAND killswitch-ui-tests -platform offscreen)
